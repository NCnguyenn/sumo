+class ControlReducer(initialSettings: ControllerSettings = ControllerSettings(),

+        private set

+    private val blocked = mutableSetOf<Int>()
+    private val awaitRelease = mutableSetOf<Int>()
+    private val joystick = mutableMapOf<Int, Pair<Float, Float>>()

+    private var killed = false
+    private var caps = 0
+    private var autoApproved = false
+    private var mode = DriveMode.MANUAL
+    private var pendingMode: DriveMode? = null

+    private var unlockStart: Long? = null
+    private var unlockPending = false
+    private var nextTurnId = 0
+    private var turnId = 0
+    private var turnDuration = 0
+    private var turnActive = false

+        ControlButton.LEFT, ControlButton.RIGHT, ControlButton.JOYSTICK)

+    val state: ControlState get() {

+        val brake = connected && !killed && ControlButton.BRAKE in valid.values
+        val manual = connected && !killed && !brake && mode == DriveMode.MANUAL && pendingMode == null && !unlockPending && turnId == 0
+        val weapon = manual && (toggledWeapon || (settings.weaponMode == WeaponMode.HOLD && ControlButton.WEAPON in valid.values))
+        val boostHeld = ControlButton.BOOST in valid.values

+            if (settings.inputMode == InputMode.JOYSTICK) {

+                DriveMixer.mix(position?.first ?: 0f, position?.second ?: 0f, settings.limitPercent, boostHeld, settings.calibration)
+            } else {
+                fun axis(positive: ControlButton, negative: ControlButton) =
+                    (if (positive in valid.values) 1f else 0f) - (if (negative in valid.values) 1f else 0f)
+                DriveMixer.mix(axis(ControlButton.RIGHT, ControlButton.LEFT), axis(ControlButton.FORWARD, ControlButton.BACKWARD),
+                    settings.limitPercent, boostHeld, settings.calibration.copy(sensitivity = 1f, deadZone = 0f))
+            }
+        } else WheelPwm(0, 0)
+        return ControlState(pwm, boostHeld && pwm != WheelPwm(0, 0), weapon, brake, killed, connected, mode,
+            pendingMode, turnId, turnDuration, turnActive, unlockPending)
+    }

+    fun connectAcknowledged(capabilities: Int = 3, autoApproved: Boolean = false, receiverKilled: Boolean = false) {
+        require(capabilities and 1 != 0)
+        cancelAll()
+        caps = capabilities
+        this.autoApproved = autoApproved
+        killed = killed || receiverKilled
+        nextTurnId = 0
+        connected = true
+    }
+    fun disconnect() { cancelAll(); connected = false }
+
+    fun press(pointerId: Int, button: ControlButton) {
+        if (pointerId in pressed) return
+        pressed[pointerId] = button
+        unlockStart = null
+        unlockPending = false
+        if (!connected || killed) { blocked.add(pointerId); return }
+        if (button == ControlButton.BRAKE) {
+            invalidateExceptBrakes()
+            return
+        }
+        if (state.brake || mode != DriveMode.MANUAL || pendingMode != null) { blocked.add(pointerId); return }
+        if (turnId != 0) {
+            if (button.direction()) { endTurn(); awaitRelease.add(pointerId) }
+            blocked.add(pointerId)
+            return
+        }
+        if (button.direction()) {
+            val inputMatches = (button == ControlButton.JOYSTICK) == (settings.inputMode == InputMode.JOYSTICK)
+            val otherJoystick = pressed.any { (id, value) -> id != pointerId && id !in blocked && value == ControlButton.JOYSTICK }
+            if (!inputMatches || awaitRelease.isNotEmpty() || (button == ControlButton.JOYSTICK && otherJoystick)) {
+                blocked.add(pointerId)
+                return
+            }
+            if (button == ControlButton.JOYSTICK) joystick[pointerId] = 0f to 0f
+        }
+        if (button == ControlButton.WEAPON && settings.weaponMode == WeaponMode.TOGGLE) toggledWeapon = !toggledWeapon
+    }
+
+    fun moveJoystick(pointerId: Int, x: Float, y: Float) {
+        if (pressed[pointerId] == ControlButton.JOYSTICK && pointerId !in blocked) joystick[pointerId] = x to y
+    }
+    private fun invalidateDirections() {
+        for ((id, button) in pressed) if (button.direction() && id !in blocked) {
+            blocked.add(id)
+            awaitRelease.add(id)
+        }
+        endTurn()
+    }
+    fun release(pointerId: Int) {
+        val button = pressed.remove(pointerId) ?: return
+        val wasValid = !blocked.remove(pointerId)
+        awaitRelease.remove(pointerId)
+        joystick.remove(pointerId)
+        if (wasValid && button.direction()) invalidateDirections()
+        if (wasValid && button == ControlButton.BRAKE) invalidateExceptBrakes()
+    }
+    fun leaveControl(pointerId: Int) {
+        if (pressed[pointerId]?.direction() == true && pointerId !in blocked) invalidateDirections()
+    }
+    fun tick(): ControlState = state
+
+    private fun endTurn() { turnId = 0; turnDuration = 0; turnActive = false }
+    private fun invalidateExceptBrakes() {
+        for ((id, button) in pressed) if (button != ControlButton.BRAKE) { blocked.add(id); awaitRelease.add(id) }
+        toggledWeapon = false
+        endTurn()
+        pendingMode = null
+        mode = DriveMode.MANUAL
+        unlockPending = false
+        unlockStart = null
+    }
+    fun cancelAll() {
+        pressed.clear(); blocked.clear(); awaitRelease.clear(); joystick.clear()
+        toggledWeapon = false
+        endTurn()
+        mode = DriveMode.MANUAL
+        pendingMode = null
+        unlockPending = false
+        unlockStart = null
+    }
+    fun kill() {
+        blocked.addAll(pressed.keys)
+        awaitRelease.addAll(pressed.keys)
+        toggledWeapon = false
+        endTurn()
+        killed = true
+        mode = DriveMode.MANUAL
+        pendingMode = null
+        unlockStart = null
+        unlockPending = false
+    }
+    fun beginUnlock(): Boolean {
+        if (!connected || !killed || pressed.isNotEmpty() || unlockPending) return false
+        unlockStart = nowMs()
+        return true
+    }
+    fun unlock(): Boolean {
+        val start = unlockStart ?: return false
+        if (!connected || !killed || pressed.isNotEmpty() || unlockPending || nowMs() - start < 1000) return false
+        unlockPending = true
+        return true
+    }
+    fun acknowledgeUnlock(accepted: Boolean) {
+        if (!unlockPending || !connected || pressed.isNotEmpty()) return
+        unlockPending = false
+        unlockStart = null
+        if (accepted) { killed = false; cancelAll() }
+    }
+    fun requestMode(mode: DriveMode): Boolean {
+        if (!connected || killed || pressed.isNotEmpty() || pendingMode != null || turnId != 0 || state.weapon || this.mode == mode) return false
+        if (mode == DriveMode.AUTO && (caps and 4 == 0 || !autoApproved)) return false
+        pendingMode = mode
+        return true
+    }
+    fun acknowledgeMode(mode: DriveMode, accepted: Boolean) {
+        if (!connected || killed || pendingMode != mode) return
+        pendingMode = null
+        if (accepted) this.mode = mode
+    }
+    fun quickTurn(): Boolean {
+        if (!connected || killed || mode != DriveMode.MANUAL || pendingMode != null || pressed.isNotEmpty() ||
+            state.weapon || turnId != 0 || caps and 2 == 0 || nextTurnId == Int.MAX_VALUE) return false
+        turnId = ++nextTurnId
+        turnDuration = settings.turnDurationMs
+        turnActive = false
+        return true
+    }
+    fun acknowledgeTurn(id: Int, active: Boolean, accepted: Boolean = true) {
+        if (id != turnId || turnId == 0) return
+        // The receiver owns the timer; retransmitting the same ID must not restart it.
+        if (!accepted || !active) endTurn() else turnActive = true
+    }
+    fun setLimiter(percent: Int) { settings = settings.copy(limitPercent = percent).validated() }
+    fun applySettings(value: ControllerSettings) {
+        check(!connected && pressed.isEmpty() && turnId == 0)
+        settings = value.validated()
+    }
+}

+package com.aistudy.myapplication.control
+
+import com.aistudy.myapplication.settings.DriveCalibration
+import kotlin.math.abs
+import kotlin.math.hypot
+import kotlin.math.max
+import kotlin.math.min
+import kotlin.math.roundToInt
+
+data class WheelPwm(val left: Int, val right: Int) {
+    init { require(left in -255..255 && right in -255..255) }
+}
+
+object DriveMixer {
+    fun mix(x: Float, y: Float, limitPercent: Int, boost: Boolean,
+            calibration: DriveCalibration = DriveCalibration()): WheelPwm {
+        require(limitPercent in 30..100)
+        calibration.validated()
+        if (!x.isFinite() || !y.isFinite()) return WheelPwm(0, 0)
+        val cx = x.coerceIn(-1f, 1f).toDouble()
+        val cy = y.coerceIn(-1f, 1f).toDouble()
+        val radius = hypot(cx, cy)
+        if (radius <= calibration.deadZone) return WheelPwm(0, 0)
+        val magnitude = min(1.0, calibration.sensitivity * (min(radius, 1.0) - calibration.deadZone) / (1 - calibration.deadZone))
+        val turn = cx / radius * magnitude * calibration.turnPercent / 100
+        val forward = cy / radius * magnitude
+        val scale = max(1.0, max(abs(forward + turn), abs(forward - turn)))
+        val left = (forward + turn) / scale * calibration.leftGain
+        val right = (forward - turn) / scale * calibration.rightGain
+        val peak = max(abs(left), abs(right))
+        if (peak == 0.0) return WheelPwm(0, 0)
+        val multiplier = if (boost) 255 / peak else 255 * limitPercent / 100.0
+        fun pwm(value: Double): Int {
+            val rounded = abs(value * multiplier).roundToInt().coerceIn(0, 255)
+            return if (value < 0) -rounded else rounded
+        }
+        return WheelPwm(pwm(left), pwm(right))
+    }

+package com.aistudy.myapplication.protocol
+
+import com.aistudy.myapplication.control.*
+import com.aistudy.myapplication.settings.ControllerSettings

+sealed interface Payload {
+    data class Drive(val pwm:WheelPwm=WheelPwm(0,0),val boost:Boolean=false,val weapon:Boolean=false,val brake:Boolean=false):Payload
+    data object Unlock:Payload
+    data object Close:Payload
+    data class Configure(val id:Int,val homeDeg:Int,val activeDeg:Int):Payload
+}
+data class ControlFrame(val sid:String,val seq:Int,val kill:Boolean=false,val mode:DriveMode=DriveMode.MANUAL,
+    val turnId:Int=0,val durationMs:Int=0,val payload:Payload=Payload.Drive())
+sealed interface RobotReply {
+    data class Offer(val nonce:String,val sid:String,val capabilities:Int):RobotReply
+    data class Busy(val nonce:String):RobotReply
+    data class Ack(val sid:String,val seq:Int,val result:String,val state:ControlState):RobotReply
+    data class ConfigAck(val sid:String,val seq:Int,val configId:Int,val homeDeg:Int,val activeDeg:Int,val result:String):RobotReply
+}
+object CommandCodec {
+    private val results = setOf("OK", "LOCKED", "BAD", "UNSUPPORTED", "BUSY")
+    private fun id(value:String):String = value.also { require(it.matches(Regex("[0-9a-f]{16}")) && it != "0".repeat(16)) }
+    private fun number(value:String, min:Int=0, max:Int=Int.MAX_VALUE):Int {
+        require(value.matches(Regex(if (min < 0) "-?[0-9]+" else "[0-9]+")))
+        return requireNotNull(value.toIntOrNull()).also { require(it in min..max) }
+    }
+    private fun flag(value:String):Boolean { require(value == "0" || value == "1"); return value == "1" }
+    private fun Boolean.bit() = if (this) 1 else 0
+    private fun text(bytes:ByteArray):String {
+        require(bytes.size in 1..256 && bytes.last() == 10.toByte())
+        require(bytes.dropLast(1).all { (it.toInt() and 255) in 32..126 })
+        return String(bytes, Charsets.US_ASCII).dropLast(1)
+    }
+    private fun wire(value:String):ByteArray = value.toByteArray(Charsets.US_ASCII).also { text(it) }
+    fun encodeHello(nonce:String):ByteArray = wire("H1:${id(nonce)}\n")
+    fun encodeDrive(pwm:WheelPwm,boost:Boolean,weapon:Boolean,brake:Boolean,template:String=ControllerSettings.S_TEMPLATE):String {
+        require(template == ControllerSettings.S_TEMPLATE || template == ControllerSettings.D_TEMPLATE)
+        val verb = if (template == ControllerSettings.D_TEMPLATE) "D" else "S"
+        return "$verb:${pwm.left},${pwm.right},${boost.bit()},${weapon.bit()},${brake.bit()}\n"
+    }
+    fun encodeControl(frame:ControlFrame,template:String=ControllerSettings.S_TEMPLATE):ByteArray {
+        id(frame.sid)
+        require(frame.seq > 0 && frame.turnId >= 0)
+        require((frame.turnId == 0 && frame.durationMs == 0) || (frame.turnId > 0 && frame.durationMs in 1..2000))
+        val neutralMetadata = !frame.kill && frame.mode == DriveMode.MANUAL && frame.turnId == 0
+        val body = when (val p = frame.payload) {
+            is Payload.Drive -> {
+                val zero = p.pwm == WheelPwm(0, 0)
+                if (frame.kill) require(frame.mode == DriveMode.MANUAL && frame.turnId == 0 && zero && !p.boost && !p.weapon && !p.brake)
+                if (p.brake) require(neutralMetadata && zero && !p.boost && !p.weapon)
+                if (frame.mode == DriveMode.AUTO || frame.turnId > 0) require(!frame.kill && zero && !p.boost && !p.weapon && !p.brake)
+                if (frame.mode == DriveMode.AUTO) require(frame.turnId == 0)
+                if (frame.turnId > 0) require(frame.mode == DriveMode.MANUAL)
+                require(!p.boost || !zero)
+                encodeDrive(p.pwm,p.boost,p.weapon,p.brake,template).dropLast(1)
+            }
+            Payload.Unlock -> { require(neutralMetadata); "U" }
+            Payload.Close -> { require(neutralMetadata); "C" }
+            is Payload.Configure -> {
+                require(neutralMetadata && p.id > 0 && p.homeDeg in 0..180 && p.activeDeg in 0..180)
+                "Q:${p.id},${p.homeDeg},${p.activeDeg}"
+            }
+        }
+        return wire("V1:${frame.sid},${frame.seq},${frame.kill.bit()},${if(frame.mode==DriveMode.AUTO) 1 else 0},${frame.turnId},${frame.durationMs}|$body\n")
+    }
+    fun decodeReply(bytes:ByteArray):RobotReply {
+        val raw=text(bytes).split(':',limit=2)
+        require(raw.size==2)
+        val p=raw[1].split(',')
+        return when(raw[0]) {
+            "H1" -> {
+                require(p.size==3)
+                val caps=number(p[2],1)
+                require(caps and 1 != 0)
+                RobotReply.Offer(id(p[0]),id(p[1]),caps)
+            }
+            "N1" -> { require(p.size==2 && p[1]=="BUSY"); RobotReply.Busy(id(p[0])) }
+            "Q1" -> {
+                require(p.size==6 && p[5] in results)
+                RobotReply.ConfigAck(id(p[0]),number(p[1],1),number(p[2],1),number(p[3],0,180),number(p[4],0,180),p[5])
+            }
+            "A1" -> {
+                require(p.size==12 && p[2] in results)
+                val pwm=WheelPwm(number(p[3],-255,255),number(p[4],-255,255))
+                val b=flag(p[5]); val w=flag(p[6]); val k=flag(p[7]); val x=flag(p[8])
+                val mode=if(flag(p[9])) DriveMode.AUTO else DriveMode.MANUAL
+                val turn=number(p[10]); val active=flag(p[11]); val zero=pwm==WheelPwm(0,0)
+                require(!b || !zero)
+                if(x) require(zero && !b && !w && !k && mode==DriveMode.MANUAL && turn==0 && !active)
+                if(k) require(zero && !b && !w && !x && mode==DriveMode.MANUAL && turn==0 && !active)
+                if(active) require(turn>0 && pwm==WheelPwm(-255,255) && !b && !w && !k && !x && mode==DriveMode.MANUAL)
+                if(turn>0 && !active) require(zero && !b && !w && !k && !x && mode==DriveMode.MANUAL)
+                RobotReply.Ack(id(p[0]),number(p[1],1),p[2],ControlState(pwm=pwm,boost=b,weapon=w,brake=k,killed=x,mode=mode,turnId=turn,turnActive=active))
+            }
+            else -> throw IllegalArgumentException("Unsupported SUMO-SIM/1 reply")
+        }
+    }
+}

+class ReplyTracker(private val host:String,private val port:Int) {
+    init { ControllerSettings(host=host,port=port).validated() }
+    var sid:String? = null
+        private set
+    var capabilities:Int = 0
+        private set
+    var online:Boolean = false
+        private set
+    var lastRttMs:Long? = null
+        private set
+    private var nonce:String?=null
+    private var startedAt:Long?=null
+    private var lastProgress:Long?=null
+    private var lastSent=0
+    private var lastAck=0
+    private var intentFloor=1
+    private var lastIntent:ControlFrame?=null
+    private data class Sent(val frame:ControlFrame,val at:Long)
+    private val pending=linkedMapOf<Int,Sent>()
+    private fun clear() {
+        sid=null; capabilities=0; online=false; lastRttMs=null
+        nonce=null; startedAt=null; lastProgress=null; lastSent=0; lastAck=0
+        intentFloor=1; lastIntent=null; pending.clear()
+    }
+    fun begin(nonce:String,nowMs:Long) {
+        CommandCodec.encodeHello(nonce)
+        clear(); this.nonce=nonce; startedAt=nowMs
+    }
+    fun recordSent(frame:ControlFrame,nowMs:Long) {
+        tick(nowMs)
+        require(sid!=null && frame.sid==sid && startedAt!=null && nowMs>=startedAt!!)
+        CommandCodec.encodeControl(frame)
+        if(!online) require(frame==ControlFrame(sid!!,1))
+        val existing=pending[frame.seq]
+        if(existing!=null) { require(existing.frame==frame); return }
+        require(frame.seq>lastSent)
+        lastSent=frame.seq
+        pending[frame.seq]=Sent(frame,nowMs)
+        if(frame.payload !is Payload.Configure) {
+            val intent=frame.copy(seq=1)
+            if(intent!=lastIntent) { intentFloor=frame.seq; lastIntent=intent }
+        }
+        while(pending.size>256) pending.remove(pending.keys.first())
+    }
+    private fun confirms(frame:ControlFrame,reply:RobotReply.Ack):Boolean {
+        val s=reply.state
+        val stopped=s.pwm==WheelPwm(0,0) && !s.boost && !s.weapon && !s.brake &&
+            s.mode==DriveMode.MANUAL && s.turnId==0 && !s.turnActive
+        return when(val p=frame.payload) {
+            Payload.Unlock -> stopped && !s.killed
+            Payload.Close -> stopped
+            is Payload.Configure -> false
+            is Payload.Drive -> when {
+                frame.kill -> stopped && s.killed
+                frame.turnId>0 -> !s.killed && !s.brake && (s.turnId==frame.turnId || (s.turnId==0 && stopped))
+                frame.mode==DriveMode.AUTO -> s.mode==DriveMode.AUTO && s.pwm==WheelPwm(0,0) &&
+                    !s.boost && !s.weapon && !s.brake && !s.killed && s.turnId==0 && !s.turnActive
+                s.killed -> p==Payload.Drive() && stopped
+                else -> s.pwm==p.pwm && s.boost==p.boost && s.weapon==p.weapon && s.brake==p.brake &&
+                    s.mode==DriveMode.MANUAL && s.turnId==0 && !s.turnActive
+            }
+        }
+    }
+    fun accept(bytes:ByteArray,fromHost:String,fromPort:Int,nowMs:Long):RobotReply? {
+        tick(nowMs)
+        val start=startedAt ?: return null
+        if(fromHost!=host || fromPort!=port || nowMs<start) return null
+        val reply=try { CommandCodec.decodeReply(bytes) } catch(_:IllegalArgumentException) { return null }
+        if(reply is RobotReply.Offer) {
+            if(online || reply.nonce!=nonce || (sid!=null && sid!=reply.sid)) return null
+            sid=reply.sid; capabilities=reply.capabilities
+            return reply
+        }
+        val replySid:String
+        val seq:Int
+        when(reply) {
+            is RobotReply.Ack -> { replySid=reply.sid; seq=reply.seq }
+            is RobotReply.ConfigAck -> { replySid=reply.sid; seq=reply.seq }
+            else -> return null
+        }
+        if(replySid!=sid || seq<=lastAck) return null
+        val sent=pending[seq] ?: return null
+        val age=nowMs-sent.at
+        if(age !in 0..299) return null
+        if(reply is RobotReply.ConfigAck) {
+            val config=sent.frame.payload as? Payload.Configure ?: return null
+            if(!online || config.id!=reply.configId || config.homeDeg!=reply.homeDeg || config.activeDeg!=reply.activeDeg) return null
+        } else if(reply is RobotReply.Ack) {
+            if(seq<intentFloor) return null
+            if(reply.result=="OK" && !confirms(sent.frame,reply)) return null
+            if(sent.frame.payload is Payload.Configure) {
+                if(reply.result=="OK" || !online) return null
+            } else {
+                if(!online) {
+                    val s=reply.state
+                    if(seq!=1 || reply.result!="OK" || s.pwm!=WheelPwm(0,0) || s.boost || s.weapon || s.brake ||
+                        s.mode!=DriveMode.MANUAL || s.turnId!=0 || s.turnActive) return null
+                }
+                online=true; lastProgress=nowMs; lastRttMs=age
+            }
+        }
+        lastAck=seq
+        pending.keys.removeAll { it<=seq }
+        if(reply is RobotReply.Ack && sent.frame.payload==Payload.Close && reply.result=="OK") clear()
+        return reply
+    }
+    fun tick(nowMs:Long) {
+        val start=startedAt ?: return
+        val progress=lastProgress
+        if((progress==null && nowMs-start>=1000) || (progress!=null && nowMs-progress>=300)) clear()
+    }

+package com.aistudy.myapplication.settings

+enum class WeaponMode { HOLD, TOGGLE }
+enum class InputMode { D_PAD, JOYSTICK }

+data class DriveCalibration(
+    val sensitivity: Float = 1f, val deadZone: Float = 0f,
+    val leftGain: Float = 1f, val rightGain: Float = 1f, val turnPercent: Int = 100
+) {
+    fun validated(): DriveCalibration = apply {
+        require(sensitivity.isFinite() && sensitivity in 0.25f..2f)

+"""SUMO-SIM/1 receiver for local tests, independent of the Android encoder."""
+import argparse
+import csv
+import heapq
+import ipaddress
+import json
+import re
+import secrets
+import socket
+import time
+import sys
+import unittest
+import xml.etree.ElementTree as ET
+from pathlib import Path
+
+
+class Receiver:
+    def __init__(self, clock=None, sid_factory=None, caps=3, log=None):
+        if not 0 < caps <= 2147483647 or not caps & 1: raise ValueError("caps must include state capability")
+        self.clock=clock or (lambda: time.monotonic_ns() // 1_000_000)
+        self.sid_factory=sid_factory or (lambda: secrets.token_hex(8))
+        self.log=log; self.caps=caps; self.sid=None; self.active=False
+        self.last_valid=None; self.last_seq=0; self.turn_count=0; self.config_pair=(0,90)
+        self.state=dict(L=0,R=0,B=0,W=0,K=0,X=0,mode=0,turnId=0,turnActive=0)
+        self.peer=None; self.nonce=None; self.offer_deadline=None; self.previous_sid=None
+        self.acks={}; self.highest_turn=0; self.turn_duration=0; self.turn_deadline=None
+        self.config_id=0
+
+    def emit(self,event,data=b"",peer=None,reason="",accepted=None):
+        if self.log:
+            self.log(dict(at_ms=self.clock(),event=event,sid=self.sid,seq=self.last_seq,
+                          peer=peer or self.peer,payload_hex=data.hex(),reason=reason,accepted=accepted,
+                          applied=dict(self.state),config=list(self.config_pair),active=self.active))
+
+    @staticmethod
+    def ident(value):
+        if not re.fullmatch(r"[0-9a-f]{16}",value) or value=="0"*16: raise ValueError("invalid identity")
+        return value
+
+    @staticmethod
+    def number(value,low=0,high=2147483647):
+        if not re.fullmatch(r"-?[0-9]+" if low<0 else r"[0-9]+",value): raise ValueError("invalid integer")
+        result=int(value)
+        if not low<=result<=high: raise ValueError("integer out of range")
+        return result
+
+    @staticmethod
+    def flag(value):
+        if value not in ("0","1"): raise ValueError("flag must be 0 or 1")
+        return int(value)
+
+    @staticmethod
+    def wire_text(data):
+        if not 1<=len(data)<=256 or data[-1:]!=b"\n" or any(not 32<=c<=126 for c in data[:-1]):
+            raise ValueError("ASCII frame requires exactly one final LF, maximum 256 bytes")
+        return data[:-1].decode("ascii")
+
+    def stop(self,brake=0,kill=None):
+        if kill is None: kill=self.state["X"]
+        self.state.update(L=0,R=0,B=0,W=0,K=brake,X=kill,mode=0,turnId=0,turnActive=0)
+        self.turn_deadline=None
+
+    def neutral(self):
+        return all(self.state[k]==0 for k in ("L","R","B","W","K","turnActive"))
+
+    def retire(self,reason):
+        self.emit("retire",reason=reason)
+        self.previous_sid=self.sid
+        self.sid=None; self.peer=None; self.nonce=None; self.active=False
+        self.offer_deadline=None; self.last_valid=None; self.last_seq=0; self.acks.clear()
+        self.highest_turn=0; self.turn_duration=0; self.config_id=0
+
+    def advance(self,now):
+        if self.active and now-self.last_valid>=300:
+            self.stop(); self.emit("watchdog",reason="300ms without fresh applied control"); self.retire("watchdog")
+        elif self.turn_deadline is not None and now>=self.turn_deadline:
+            self.state.update(L=0,R=0,B=0,turnActive=0); self.turn_deadline=None
+            self.emit("macro-complete")
+        if not self.active and self.sid is not None and now>=self.offer_deadline:
+            self.retire("offer-expired")
+
+    def tick(self): self.advance(self.clock())
+
+    def ack(self,seq,result):
+        values=",".join(str(self.state[k]) for k in ("L","R","B","W","K","X","mode","turnId","turnActive"))
+        return f"A1:{self.sid},{seq},{result},{values}\n".encode("ascii")
+
+    def cache(self,seq,data,reply):
+        self.acks[seq]=(data,reply)
+        while len(self.acks)>256: self.acks.pop(next(iter(self.acks)))
+
+    def hello(self,text,data,peer,now):
+        nonce=self.ident(text[3:])
+        if self.active or (self.sid is not None and (self.peer!=peer or self.nonce!=nonce)):
+            self.emit("reject",data,peer,"session-busy",False)
+            return f"N1:{nonce},BUSY\n".encode("ascii")
+        if self.sid is None:
+            new_sid=self.ident(self.sid_factory())
+            if new_sid==self.previous_sid: raise ValueError("SID must be fresh")
+            self.sid=new_sid; self.peer=peer; self.nonce=nonce; self.offer_deadline=now+1000
+            self.stop(); self.emit("offer",data,peer)
+        return f"H1:{nonce},{self.sid},{self.caps}\n".encode("ascii")
+
+    def drive(self,x,mode,tid,duration,body,now):
+        p=body[2:].split(',')
+        if len(p)!=5: raise ValueError("drive field count")
+        left,right=self.number(p[0],-255,255),self.number(p[1],-255,255)
+        boost,weapon,brake=(self.flag(v) for v in p[2:])
+        zero=(left,right,boost,weapon,brake)==(0,0,0,0,0)
+        if boost and left==right==0: raise ValueError("boost without direction")
+        if x and (mode or tid or not zero): raise ValueError("contradictory Kill")
+        if brake and (x or mode or tid or (left,right,boost,weapon)!=(0,0,0,0)): raise ValueError("contradictory Brake")
+        if (mode or tid) and (x or not zero): raise ValueError("contradictory mode or turn")
+        if mode and tid: raise ValueError("AUTO and macro cannot coexist")
+        if x: self.stop(kill=1); return "OK"
+        if self.state["X"]: return "OK" if zero and mode==tid==0 else "LOCKED"
+        if brake: self.stop(brake=1); return "OK"
+        if mode:
+            if not self.caps & 4: return "UNSUPPORTED"
+            if not self.neutral(): return "BUSY"
+            self.stop(); self.state["mode"]=1; return "OK"
+        if self.state["mode"]:
+            if not zero or tid: return "BUSY"
+            self.stop(); return "OK"
+        if tid:
+            if not self.caps & 2: return "UNSUPPORTED"
+            if tid<self.highest_turn: return "BAD"
+            if tid==self.highest_turn: return "OK" if duration==self.turn_duration else "BAD"
+            if not self.neutral(): return "BUSY"
+            self.highest_turn=tid; self.turn_duration=duration
+            self.stop(); self.state.update(L=-255,R=255,turnId=tid,turnActive=1)
+            self.turn_deadline=now+duration; self.turn_count+=1; self.emit("macro-start")
+            return "OK"
+        if self.state["turnActive"] and not zero: return "BUSY"
+        self.turn_deadline=None
+        self.state.update(L=left,R=right,B=boost,W=weapon,K=0,mode=0,turnId=0,turnActive=0)
+        return "OK"

+    def configure(self,seq,body):
+        p=body[2:].split(',')
+        if len(p)!=3: raise ValueError("config field count")
+        cfg=self.number(p[0],1); home=self.number(p[1],0,180); active=self.number(p[2],0,180)
+        if not self.caps & 8: result="UNSUPPORTED"
+        elif self.state["X"]: result="LOCKED"
+        elif not self.neutral() or self.state["mode"]: result="BUSY"
+        elif cfg<self.config_id or (cfg==self.config_id and (home,active)!=self.config_pair): result="BAD"
+        else:
+            self.config_pair=(home,active); self.config_id=cfg; result="OK"
+            self.emit("config-applied")
+        return f"Q1:{self.sid},{seq},{cfg},{home},{active},{result}\n".encode("ascii"),result

+    def handle(self,data,peer):
+        now=self.clock(); self.advance(now)
+        try:
+            text=self.wire_text(data)
+            if text.startswith("H1:"): return self.hello(text,data,peer,now)
+            if not text.startswith("V1:") or text.count('|')!=1: raise ValueError("V1 envelope required")
+            envelope,body=text[3:].split('|')
+            p=envelope.split(',')
+            if len(p)!=6: raise ValueError("envelope field count")
+            sid=self.ident(p[0]); seq=self.number(p[1],1); x=self.flag(p[2]); mode=self.flag(p[3])
+            tid=self.number(p[4]); duration=self.number(p[5],0,2000)
+            if (tid==0 and duration!=0) or (tid>0 and duration==0): raise ValueError("turn metadata mismatch")
+        except ValueError as error:
+            self.emit("reject",data,peer,str(error),False); return None
+        if sid!=self.sid or peer!=self.peer:
+            self.emit("reject",data,peer,"wrong-peer-or-retired-sid",False); return None
+        if seq<=self.last_seq:

+            self.emit("reject",data,peer,"duplicate-or-old-seq",False)
+            return cached[1] if cached and cached[0]==data else None
+        self.last_seq=seq
+        if not self.active:

+                self.active=True; self.last_valid=now; self.offer_deadline=None; result="OK"
+            else: result="BAD"
+            reply=self.ack(seq,result); self.cache(seq,data,reply)
+            self.emit("accept" if result=="OK" else "reject",data,peer,result,result=="OK")
+            return reply
+        control=True
+        try:
+            if body.startswith(("S:","D:")):

+            else:
+                if (x,mode,tid,duration)!=(0,0,0,0): raise ValueError("non-neutral control envelope")
+                if body=="U":

+                    reply=self.ack(seq,result)
+                elif body=="C":
+                    self.stop(); reply=self.ack(seq,"OK")
+                    self.emit("accept",data,peer,"closed",True); self.retire("close")
+                    return reply
+                elif body.startswith("Q:"):
+                    control=False; reply,result=self.configure(seq,body)
+                else: raise ValueError("unsupported payload")
+        except ValueError:
+            result="BAD"; reply=self.ack(seq,result)
+        if result=="OK" and control: self.last_valid=now
+        self.cache(seq,data,reply)
+        self.emit("accept" if result=="OK" else "reject",data,peer,result,result=="OK")
+        return reply

+class Scenario:
+    def __init__(self, receiver, name="normal", fault_after=2, delay_ms=150, delay_target="rx"):
+        if name not in ("normal","drop-rx","drop-reply","delay","duplicate","reorder","malformed"): raise ValueError("unknown scenario")
+        if fault_after<0 or delay_ms<0 or delay_target not in ("rx","reply"): raise ValueError("invalid scenario arguments")
+        self.receiver=receiver; self.name=name; self.fault_after=fault_after; self.delay_ms=delay_ms; self.delay_target=delay_target
+        self.count=0; self.order=0; self.queue=[]; self.held=None; self.malformed_count=0
+
+    def later(self,delay,kind,data,peer):
+        self.order+=1; heapq.heappush(self.queue,(self.receiver.clock()+delay,self.order,kind,data,peer))
+
+    def deliver(self,data,peer):
+        reply=self.receiver.handle(data,peer)
+        return [] if reply is None else [(peer,reply)]
+
+    @staticmethod
+    def motion(data):
+        try:
+            body=data.split(b'|',1)[1]
+            if not body.startswith((b'S:',b'D:')): return None
+            p=body[2:].split(b',')
+            return int(p[0])!=0 or int(p[1])!=0
+        except (IndexError,ValueError): return None
+
+    def receive(self,data,peer):
+        self.receiver.tick()
+        is_control=data.startswith(b'V1:')
+        if is_control: self.count+=1
+        if not is_control or self.count<=self.fault_after or self.name=="normal": return self.deliver(data,peer)
+        self.receiver.emit("scenario",data,peer,self.name)
+        if self.name=="drop-rx": return []
+        if self.name=="delay" and self.delay_target=="rx":
+            self.later(self.delay_ms,"rx",data,peer); return []
+        if self.name=="reorder":
+            motion=self.motion(data)
+            if motion is True and self.held is None:
+                self.held=(data,peer); return []
+            replies=self.deliver(data,peer)
+            if motion is False and self.held is not None:
+                old,old_peer=self.held; self.held=None; self.later(100,"rx",old,old_peer)
+            return replies
+        replies=self.deliver(data,peer)
+        if self.name=="drop-reply": return []
+        if self.name=="delay":
+            for target,reply in replies: self.later(self.delay_ms,"reply",reply,target)
+            return []
+        if self.name=="duplicate": return replies+self.deliver(data,peer)
+        if self.name=="malformed":
+            mutated=[]
+            for target,reply in replies:
+                prefix,fields=reply.decode('ascii').rstrip('\n').split(':',1); p=fields.split(',')
+                kind=self.malformed_count%3; self.malformed_count+=1
+                if kind==0: p=p[:-1]
+                elif kind==1: p[0]='f'*16
+                else: p[1]=str(int(p[1])+1)
+                changed=(prefix+':'+','.join(p)+'\n').encode('ascii')
+                self.receiver.emit("scenario-reply",changed,target,"malformed-"+str(kind))
+                mutated.append((target,changed))
+            return mutated
+        return replies
+
+    def poll(self):
+        self.receiver.tick(); replies=[]; now=self.receiver.clock()
+        while self.queue and self.queue[0][0]<=now:
+            _,_,kind,data,peer=heapq.heappop(self.queue)
+            if kind=="rx": replies.extend(self.deliver(data,peer))
+            else: replies.append((peer,data))
+        return replies
